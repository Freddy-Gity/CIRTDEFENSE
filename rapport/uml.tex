%%%===========================================================================
%%% uml.tex — outillage de dessin UML en TikZ pur
%%%
%%% Choix assume : aucune dependance a tikz-uml / pgf-umlcd / pgf-umlsd. Les
%%% diagrammes sont traces avec les seules bibliotheques PGF du noyau, deja
%%% presentes dans toute distribution TeX Live et sur Overleaf. Le rapport
%%% compile donc partout ou le modele d'origine compile, sans installation.
%%%
%%% Conventions de notation retenues (UML 2.5.1, OMG formal/2017-12-05) :
%%%   — generalisation      : trait plein,   pointe triangulaire creuse
%%%   — realisation         : trait pointille, pointe triangulaire creuse
%%%   — composition         : losange plein cote tout
%%%   — agregation          : losange creux cote tout
%%%   — association         : trait plein, multiplicites aux deux extremites
%%%   — dependance          : trait pointille, pointe ouverte
%%%===========================================================================

%%% -------------------------------------------------------------- classeurs
% Un classeur UML se dessine avec `rectangle split` : un compartiment par
% partie. On fixe la largeur pour que les diagrammes restent alignes.
\tikzset{
  uml/.style={
    draw=umlcadre, line width=0.5pt, font=\scriptsize,
    rectangle split, rectangle split draw splits=true,
    text width=#1, align=left, inner sep=3pt,
    rectangle split part fill={umltete,umlcorps,umlcorps},
  },
  uml/.default=3.5cm,
  umlclasse/.style={uml=#1, rectangle split parts=3},
  umlclasse/.default=3.5cm,
  umlcompact/.style={uml=#1, rectangle split parts=2},
  umlcompact/.default=3.5cm,
  umlsimple/.style={uml=#1, rectangle split parts=1},
  umlsimple/.default=3.5cm,
  umlaccentue/.style={draw=umlaccent, line width=1pt},
  % Titre d'un classeur : nom centre, stereotype eventuel au-dessus.
  umlnom/.style={font=\scriptsize\bfseries},
}

%%% ------------------------------------------------------- environnement
%%% babel-french rend « ; : ! ? » actifs pour l'espacement typographique.
%%% Ces caracteres servent aussi de separateurs dans la syntaxe de PGF/TikZ
%%% (cles, etiquettes de position, chemins). Tout diagramme est donc trace
%%% a l'interieur de cet environnement, qui suspend les raccourcis le temps
%%% du dessin et les retablit ensuite.
\newenvironment{diagramme}[1][]
  {\shorthandoff{;:!?}\begin{tikzpicture}[#1]}
  {\end{tikzpicture}\shorthandon{;:!?}}

%%% ------------------------------------------- primitives de sequence
%%% Un diagramme de sequence se construit en coordonnees explicites :
%%% l'axe des x porte les lignes de vie, l'axe des y le temps qui descend.
%%% \seqvie{nom}{x}{y bas}{libelle} : tete de ligne de vie + trait pointille
\newcommand{\seqvie}[4]{%
  \node (#1) at (#2,0) [uml=2.15cm, rectangle split parts=1,
                        align=center, font=\tiny] {#4};
  \draw[umlgris, dashed] (#1.south) -- (#2,#3);%
}
%%% \seqact{x}{y haut}{y bas} : barre d'activation
\newcommand{\seqact}[3]{%
  \draw[activation] (#1-0.09,#2) rectangle (#1+0.09,#3);%
}
%%% \seqmsg{x depart}{x arrivee}{y}{libelle} : message synchrone
\newcommand{\seqmsg}[4]{%
  \draw[message] (#1,#3) -- (#2,#3)
    node[midway, above=0.3pt, font=\tiny, align=center] {#4};%
}
%%% \seqrep{x depart}{x arrivee}{y}{libelle} : message de retour
\newcommand{\seqrep}[4]{%
  \draw[retour] (#1,#3) -- (#2,#3)
    node[midway, above=0.3pt, font=\tiny, align=center] {#4};%
}
%%% \seqauto{x}{y}{libelle} : message reflexif
\newcommand{\seqauto}[3]{%
  \draw[message] (#1+0.09,#2) -- (#1+0.8,#2) -- (#1+0.8,#2-0.4) -- (#1+0.11,#2-0.4);
  \node[font=\tiny, anchor=west, align=left] at (#1+0.9,#2-0.2) {#3};%
}
%%% \seqfrag{x1}{y haut}{x2}{y bas}{operateur}{garde} : fragment combine
\newcommand{\seqfrag}[6]{%
  \draw[cadrefrag] (#1,#2) rectangle (#3,#4);
  \node[etiquettefrag] at (#1,#2) {#5};
  \node[font=\tiny, anchor=north west] at (#1+0.95,#2) {[#6]};%
}
%%% \seqsep{x1}{x2}{y}{libelle} : separateur d'alternative dans un fragment
\newcommand{\seqsep}[4]{%
  \draw[cadrefrag, dashed] (#1,#3) -- (#2,#3);
  \node[font=\tiny, anchor=north west] at (#1+0.15,#3) {[#4]};%
}
%%% \seqfin{x}{y} : marque de fin de ligne de vie (destruction)
\newcommand{\seqfin}[2]{%
  \draw[umlcadre, line width=0.7pt] (#1-0.18,#2+0.18) -- (#1+0.18,#2-0.18);
  \draw[umlcadre, line width=0.7pt] (#1-0.18,#2-0.18) -- (#1+0.18,#2+0.18);%
}

%%% ------------------------------------------------ ajustement des figures
%%% Un diagramme trace en coordonnees absolues ne connait pas la largeur de
%%% la page. Ces deux commandes le ramenent dans la zone imprimable sans
%%% jamais l'agrandir : un petit diagramme garde sa taille naturelle.
%%%
%%% Elles n'emploient pas adjustbox : celui-ci charge trimclip, qui redefinit
%%% \clipbox deja pose par pstricks (charge par psvectorian dans le preambule
%%% du modele). Le facteur d'echelle est donc calcule a la main avec pgfmath,
%%% deja disponible puisque TikZ est charge.
\newsavebox{\boitediagramme}
\newcommand{\ajusterdans}[3]{%
  \sbox{\boitediagramme}{#3}%
  \pgfmathsetmacro{\facteurdiagramme}{%
    min(1,
        (#1)/(\wd\boitediagramme + 0.01pt),
        (#2)/(\ht\boitediagramme + \dp\boitediagramme + 0.01pt))}%
  \scalebox{\facteurdiagramme}{\usebox{\boitediagramme}}%
}
\newcommand{\ajusterportrait}[1]{\ajusterdans{\linewidth}{0.88\textheight}{#1}}
\newcommand{\ajusterpaysage}[1]{\ajusterdans{\linewidth}{0.92\textwidth}{#1}}
